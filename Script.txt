local function devePreservar(nome)
    local n = nome:lower()
    for _, termo in ipairs(preservar) do
        if n:find(termo) then return true end
    end
    return false
end

local function esconderOuDestruirUI(parent)
    if not parent then return end
    for _, gui in ipairs(parent:GetChildren()) do
        if gui:IsA("ScreenGui") then
            local nome = gui.Name
            if not devePreservar(nome) and ehHubSecundario(nome) then
                pcall(function()
                    gui.Enabled = false
                    gui:Destroy()
                end)
            end
        end
    end
end

local function getSpawnPosition()
    local spawnLocation = Workspace:FindFirstChildOfClass("SpawnLocation")
    if spawnLocation then
        return spawnLocation.Position + Vector3.new(0, 5, 0)
    end
    
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d.Name == "Spawn" and d:IsA("BasePart") then
            return d.Position + Vector3.new(0, 5, 0)
        end
        if d.Name == "SpawnLocation" and d:IsA("BasePart") then
            return d.Position + Vector3.new(0, 5, 0)
        end
    end
    
    return Vector3.new(0, 50, 0)
end

local function isSmartPrompt(prompt)
    if not prompt or not prompt.Parent then return false end
    if prompt.Name == "CarryAreaEgg" then return true end
    if prompt.Parent.Name == "SmartPromptPart" then return true end
    local p, depth = prompt.Parent, 0
    while p and p ~= Workspace and depth < 15 do
        if p.Name == "SmartPromptPart" then return true end
        p = p.Parent
        depth += 1
    end
    return false
end

local function stopTpLoop()
    if tpLoopConn then
        tpLoopConn:Disconnect()
        tpLoopConn = nil
    end
end

local function startTpLoop()
    stopTpLoop()
    tpCachedPos = COORDENADA_VIVO
    if not tpCachedPos then return end

    tpLoopConn = RunService.Heartbeat:Connect(function()
        if not tpEnabled then
            stopTpLoop()
            return
        end
        if voandoAgora then return end
        if personagemMorreu then return end
        
        local char = player.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if hrp and tpCachedPos then
            hrp.CFrame = CFrame.new(tpCachedPos) * (hrp.CFrame - hrp.CFrame.Position)
        end
    end)
end

local function doTeleport(withLoop)
    if teleGuiadoAtivo then
        pararTeleGuiado()
    end

    local destino = COORDENADA_VIVO
    if not destino then return end

    local char = player.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    hrp.CFrame = CFrame.new(destino) * (hrp.CFrame - hrp.CFrame.Position)

    if withLoop then
        startTpLoop()
    end
end

local function tryTeleport(prompt, triggeringPlayer)
    if not tpEnabled then return end
    if triggeringPlayer and triggeringPlayer ~= player then return end
    if not isSmartPrompt(prompt) then return end
    if personagemMorreu then return end
    if voandoAgora then return end
    doTeleport(true)
end

local watched = setmetatable({}, {__mode = "k"})
local function watchPrompt(prompt)
    if watched[prompt] then return end
    watched[prompt] = true
    prompt.Triggered:Connect(function(triggeringPlayer)
        tryTeleport(prompt, triggeringPlayer)
    end)
end

-- FUNÇÃO: Encontrar ProximityPrompt perto de uma posição
-- ============================================================
local function encontrarPromptPerto(posicao, raio)
    raio = raio or 25
    local melhorPrompt = nil
    local menorDistancia = math.huge
    
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") then
            local parentPart = d.Parent
            if parentPart and parentPart:IsA("BasePart") then
                local dist = (parentPart.Position - posicao).Magnitude
                if dist < raio and dist < menorDistancia then
                    menorDistancia = dist
                    melhorPrompt = d
                end
            end
        end
    end
    
    return melhorPrompt
end

-- ============================================================
-- FUNÇÃO: Pega o ovo baseado na posição do ovo
-- ============================================================
local function pegarOvoPelaPosicao(eggRecord)
    if not eggRecord or not eggRecord.BoundsCFrame then return false end
    
    local eggPos = eggRecord.BoundsCFrame.Position
    local melhorPrompt = encontrarPromptPerto(eggPos, 25)
    
    if not melhorPrompt then return false end
    
    for tentativa = 1, 5 do
        if fireproximityprompt then
            pcall(function() fireproximityprompt(melhorPrompt) end)
        end
        pcall(function() ProximityPromptSvc:InputHoldBegin(melhorPrompt, player) end)
        pcall(function() melhorPrompt:InputHoldBegin() end)
        task.wait(0.15)
    end
    
    return true
end

-- ============================================================
-- 👈 FUNÇÃO PÓS-MORTE - VAI PRA COORDENADA_MORTO (ovo dropado)
-- ============================================================
local function correrParaOvoDropadoEPegar(duracaoMax)
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    voandoAgora = true
    hum.WalkSpeed = VOO_VELOCIDADE

    local inicio = tick()
    local chegouNaCoordenada = false
    local conn
    
    conn = RunService.Heartbeat:Connect(function()
        local char2 = player.Character
        local hrp2 = char2 and char2:FindFirstChild("HumanoidRootPart")
        local hum2 = char2 and char2:FindFirstChildOfClass("Humanoid")
        if not hrp2 or not hum2 then
            conn:Disconnect()
            voandoAgora = false
            personagemMorreu = false
            return
        end

        -- Vai pra COORDENADA_MORTO (ovo dropado)
        local atual = hrp2.Position
        local direcao = (COORDENADA_MORTO - atual)
        local distancia = direcao.Magnitude

        if distancia < 3 and not chegouNaCoordenada then
            hrp2.CFrame = CFrame.new(COORDENADA_MORTO) * (hrp2.CFrame - hrp2.CFrame.Position)
            hum2:Move(Vector3.zero, false)
            hum2.WalkSpeed = 0
            chegouNaCoordenada = true
            conn:Disconnect()
            
            -- 👈 PEGA O PROMPT (10 tentativas)
            task.spawn(function()
                task.wait(0.3)
                
                for tentativa = 1, 10 do
                    local promptAlvo = encontrarPromptPerto(COORDENADA_MORTO, 25)
                    if promptAlvo then
                        if fireproximityprompt then
                            pcall(function() fireproximityprompt(promptAlvo) end)
                        end
                        pcall(function() ProximityPromptSvc:InputHoldBegin(promptAlvo, player) end)
                        pcall(function() promptAlvo:InputHoldBegin() end)
                        task.wait(0.2)
                    else
                        task.wait(0.1)
                    end
                end
                
                task.wait(0.5)
                
                -- 👈 DEPOIS → CORRE PRO SPAWN
                local spawnPos = getSpawnPosition()
                if spawnPos then
                    local c = player.Character
                    local h = c and c:FindFirstChild("HumanoidRootPart")
                    local hu = c and c:FindFirstChildOfClass("Humanoid")
                    
                    if h and hu then
                        hu.WalkSpeed = VOO_VELOCIDADE
                        
                        local inicioSpawn = tick()
                        local connSpawn
                        connSpawn = RunService.Heartbeat:Connect(function()
                            local c2 = player.Character
                            local h2 = c2 and c2:FindFirstChild("HumanoidRootPart")
                            local hu2 = c2 and c2:FindFirstChildOfClass("Humanoid")
                            if not h2 or not hu2 then
                                connSpawn:Disconnect()
                                voandoAgora = false
                                personagemMorreu = false
                                return
                            end
                            
                            local dist = (spawnPos - h2.Position).Magnitude
                            if dist < 3 then
                                h2.CFrame = CFrame.new(spawnPos) * (h2.CFrame - h2.CFrame.Position)
                                hu2.WalkSpeed = 16
                                voandoAgora = false
                                personagemMorreu = false
                                connSpawn:Disconnect()
                                return
                            end
                            
                            hu2:MoveTo(spawnPos)
                            
                            if (tick() - inicioSpawn) > 30 then
                                hu2.WalkSpeed = 16
                                voandoAgora = false
                                personagemMorreu = false
                                connSpawn:Disconnect()
                            end
                        end)
                    end
                end
            end)
            return
        end

        if not chegouNaCoordenada then
            hum2:MoveTo(COORDENADA_MORTO)
        end

        if duracaoMax and (tick() - inicio) > duracaoMax then
            hum2.WalkSpeed = 16
            voandoAgora = false
            personagemMorreu = false
            conn:Disconnect()
        end
    end)
end

local lastDeathCFrame = nil

local function hookCharacter(char)
    local humanoid = char:WaitForChild("Humanoid", 10)
    if not humanoid then return end
    humanoid.Died:Connect(function()
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then
            lastDeathCFrame = hrp.CFrame
        end
        personagemMorreu = true
        stopTpLoop()
    end)
end

player.CharacterAdded:Connect(function(char)
    hookCharacter(char)
    
    if personagemMorreu and tpEnabled then
        task.spawn(function()
            task.wait(RESPAWN_WAIT)
            local hrp = char:WaitForChild("HumanoidRootPart", 5)
            if not hrp then return end
            if hrp.Parent then
                correrParaOvoDropadoEPegar(20)
            end
        end)
    else
        personagemMorreu = false
    end
end)

if player.Character then
    hookCharacter(player.Character)
end

    if egg.Rarity then
        local r = egg.Rarity
        local name = type(r) == "table" and (r.DisplayName or r._id or r.Name) or tostring(r)
        return name, RARITY_SCORE_MAP[name] or 100
    end
    local cat = egg.AssetCategory or egg.Category or egg.Name
    if cat and AssetsData then
        local aInfo = (AssetsData.Directory or AssetsData)[cat]
        if aInfo and aInfo.Rarity then
            local r = aInfo.Rarity
            local name = type(r) == "table" and (r.DisplayName or r._id or r.Name) or tostring(r)
            return name, RARITY_SCORE_MAP[name] or 100
        end
    end
    local areaData = AreasData and (AreasData.Directory or AreasData) and (AreasData.Directory or AreasData)[egg.AreaId]
    local rarity = areaData and areaData.Rarity
    local rarityId = (type(rarity) == "table" and (rarity._id or rarity.DisplayName or rarity.Name)) or (type(rarity) == "string" and rarity) or "Common"
    local rInfo = (RarityData and (RarityData.Rarities or RarityData) or {})[rarityId] or {}
    local rarityDisplayName = (type(rInfo) == "table" and (rInfo.DisplayName or rInfo._id)) or (type(rarity) == "table" and rarity.DisplayName) or rarityId or "Common"
    return rarityDisplayName, RARITY_SCORE_MAP[rarityDisplayName] or 100
end

local function GetPetIcon(record)
    if not record then return nil end
    local rawIcon = record.Icon or record.PetIcon or record.ImageAsset or record.TextureId
    if rawIcon then
        return (type(rawIcon) == "number" or not string.match(tostring(rawIcon), "://")) and ("rbxassetid://" .. tostring(rawIcon)) or tostring(rawIcon)
    end
    local targetName = record.Pet or record.PetId or record.PetName or record.AssetCategory or record.Category
    if targetName then
        if PetsData then
            local pInfo = (PetsData.Directory or PetsData)[targetName]
            if pInfo and (pInfo.Icon or pInfo.Image or pInfo.AssetId) then
                local img = pInfo.Icon or pInfo.Image or pInfo.AssetId
                return (type(img) == "number" or not string.match(tostring(img), "://")) and ("rbxassetid://" .. tostring(img)) or tostring(img)
            end
        end
        if AssetsData then
            local aInfo = (AssetsData.Directory or AssetsData)[targetName]
            if aInfo and (aInfo.Icon or aInfo.Image or aInfo.AssetId) then
                local img = aInfo.Icon or aInfo.Image or aInfo.AssetId
                return (type(img) == "number" or not string.match(tostring(img), "://")) and ("rbxassetid://" .. tostring(img)) or tostring(img)
            end
        end
    end
    return nil
end

task.spawn(function()
    while task.wait(0.5) do
        pcall(function()
            if not EggState or not EggState.ReadFieldEggs then return end
            local ok, snapshot = pcall(EggState.ReadFieldEggs)
            if ok and snapshot and snapshot.Records then
                local bestEgg = nil
                local bestScore = -1

                for _, record in ipairs(snapshot.Records) do
                    if record.State == "Slot" and record.BoundsCFrame then
                        local rarityName, score = GetEggRarityInfo(record)
                        if score > bestScore then
                            bestScore = score
                            bestEgg = record
                        end
                    end
                end

                if bestEgg then
                    melhorOvoAtual = bestEgg
                    local rarityName, _ = GetEggRarityInfo(bestEgg)
                    local iconId = GetPetIcon(bestEgg)
                    
                    EggName.Text = bestEgg.AssetCategory or "Ovo Desconhecido"
                    EggRarity.Text = rarityName
                    if bestScore >= 1000 then
                        EggRarity.TextColor3 = Color3.fromRGB(255, 215, 0)
                    elseif bestScore >= 600 then
                        EggRarity.TextColor3 = TEMA.destaque
                    elseif bestScore >= 400 then
                        EggRarity.TextColor3 = Color3.fromRGB(140, 120, 220)
                    else
                        EggRarity.TextColor3 = TEMA.destaque
                    end

                    if iconId then
                        PetIcon.Image = iconId
                        PetIcon.Visible = true
                    else
                        PetIcon.Visible = false
                    end
                    EggTimer.Text = "12:00 AM"
                else
                    melhorOvoAtual = nil
                    EggName.Text = "Nenhum Ovo"
                    EggRarity.Text = "Aguardando..."
                    EggRarity.TextColor3 = TEMA.textoFraco
                    PetIcon.Visible = false
                end
            end
        end)
    end
end)

-- ============================================================
-- 👈 FUNÇÃO pararTeleGuiado CORRIGIDA
-- ============================================================
function pararTeleGuiado()
    teleGuiadoAtivo = false
    autoPegarAtivo = false
    
    -- 👈 DESCONECTA O LOOP
    if teleGuiadoConnection then
        teleGuiadoConnection:Disconnect()
        teleGuiadoConnection = nil
    end
    
    -- 👈 CANCELA O MOVIMENTO DO PERSONAGEM
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.WalkSpeed = 16
        hum:Move(Vector3.zero, false)
    end
    
    BtnTeleGuiado.Text = "TELEGUIADO: OFF"
    BtnTeleGuiado.TextColor3 = TEMA.vermelho
end

-- ============================================================
-- 👈 FUNÇÃO iniciarTeleGuiado CORRIGIDA
-- ============================================================
local function iniciarTeleGuiado()
    if not melhorOvoAtual then
        BtnTeleGuiado.Text = "NENHUM OVO"
        BtnTeleGuiado.TextColor3 = TEMA.vermelho
        task.wait(1.5)
        BtnTeleGuiado.Text = "TELEGUIADO: OFF"
        return
    end

    -- 👈 DESCONECTA QUALQUER LOOP ANTIGO
    if teleGuiadoConnection then
        teleGuiadoConnection:Disconnect()
        teleGuiadoConnection = nil
    end

    autoPegarAtivo = false
    teleGuiadoAtivo = true
    BtnTeleGuiado.Text = "TELEGUIADO: ON"
    BtnTeleGuiado.TextColor3 = TEMA.verde

    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.WalkSpeed = VELOCIDADE_TELEGUIADO
    end

    teleGuiadoConnection = RunService.Heartbeat:Connect(function()
        -- 👈 SE DESATIVADO, DESCONECTA O LOOP
        if not teleGuiadoAtivo then
            if teleGuiadoConnection then
                teleGuiadoConnection:Disconnect()
                teleGuiadoConnection = nil
            end
            return
        end
        
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        
        if not hrp or not hum then return end
        if not melhorOvoAtual or not melhorOvoAtual.BoundsCFrame then
            pararTeleGuiado()
            return
        end

        if hum.WalkSpeed ~= VELOCIDADE_TELEGUIADO then
            hum.WalkSpeed = VELOCIDADE_TELEGUIADO
        end

        local targetPos = melhorOvoAtual.BoundsCFrame.Position
        local distance = (targetPos - hrp.Position).Magnitude

        if distance > 3 then
            hum:MoveTo(targetPos)
        else
            local novaCFrame = CFrame.new(targetPos) * (hrp.CFrame - hrp.CFrame.Position)
            hrp.CFrame = novaCFrame
            
            hum:Move(Vector3.zero, false)
            hum.WalkSpeed = 0
            
            if not autoPegarAtivo then
                autoPegarAtivo = true
                
                task.spawn(function()
                    pegarOvoPelaPosicao(melhorOvoAtual)
                end)
                
                task.wait(0.8)
                pararTeleGuiado()
            end
        end
    end)
end

BtnTeleGuiado.MouseButton1Click:Connect(function()
    if teleGuiadoAtivo then
        pararTeleGuiado()
    else
        iniciarTeleGuiado()
    end
end)
